def add(self, req: Req, num_kv_heads: int) -> None:
    if self._check_if_req_exceed_kv_capacity(req):
        return

    # FAKE backend 用于测试；正常是 Mooncake/NIXL
    backend = TransferBackend.FAKE if req.bootstrap_host == FAKE_BOOTSTRAP_HOST else self.transfer_backend
    kv_sender_class = get_kv_class(backend, KVClassType.SENDER)

    dest_tp_ranks = [self.tp_rank]
    req.disagg_kv_sender = kv_sender_class(
        mgr=self.kv_manager,
        bootstrap_addr=f"{req.bootstrap_host}:{self.bootstrap_port}",
        bootstrap_room=req.bootstrap_room,
        dest_tp_ranks=dest_tp_ranks,
        pp_rank=self.pp_rank,
    )
    self._process_req(req)
    self.queue.append(req)

def _process_req(self, req: Req) -> None:
    """Set max_new_tokens = 1, so PrefillAdder memory estimation is accurate"""
    req.sampling_params.max_new_tokens = 1
关键细节：max_new_tokens=1 告诉 PrefillAdder "我只算 1 个 token"，这样它做内存预算时只预留 prompt+1 token 的 KV，不会多占空间。真正的生成由 D 完成。

PART 3深读 2：bootstrap_thread 与 pop_bootstrapped

这两个函数构成了P 和 D 的握手桥梁。bootstrap_thread 在后台线程 recv，pop_bootstrapped 在主循环 poll。

// mooncake/conn.py:1345-1422
def start_prefill_thread(self):
    def bootstrap_thread():
        while True:
            msg = self.server_socket.recv_multipart()
            room = msg[0].decode()

            if room == "None":
                # D 首次握手：注册 KVArgs（kv/aux 指针）
                sid = msg[3].decode()
                self.decode_kv_args_table[sid] = KVArgsRegisterInfo.from_zmq(msg)
                continue

            # 具体请求的 metadata
            room = int(room)
            sid = msg[3].decode()
            required_dst_info_num = int(msg[7].decode())
            if room not in self.transfer_infos:
                self.transfer_infos[room] = {}
            self.transfer_infos[room][sid] = TransferInfo.from_zmq(msg)

            if len(self.transfer_infos[room]) == required_dst_info_num:
                self.update_status(room, KVPoll.WaitingForInput)  # ★ 触发主循环推进

    threading.Thread(target=bootstrap_thread, daemon=True).start()
// prefill.py:270-352
def pop_bootstrapped(self):
    polls = poll_and_all_reduce_attn_cp_tp_group(
        [req.disagg_kv_sender for req in self.queue],
        cp_group, tp_group,
    )
    for i, (req, poll) in enumerate(zip(self.queue, polls)):
        if poll == KVPoll.Bootstrapping: continue
        elif poll == KVPoll.Failed: abort(req); continue
        # else: WaitingForInput
        req.metadata_buffer_index = self.req_to_metadata_buffer_idx_allocator.alloc()
        num_pages = kv_to_page_num(len(req.origin_input_ids), page_size)
        req.disagg_kv_sender.init(num_pages, req.metadata_buffer_index)
        bootstrapped.append(req)
    return bootstrapped
为什么要跨 TP group all-reduce poll：TP=8 的 P 有 8 个 rank。它们必须看到一致的状态，否则一部分 rank 进入 forward 一部分还在等，会死锁。all-reduce 取 min 保证所有 rank 看到相同 transition。

PART 4深读 3：send_kv_chunk 和 transfer_worker

请求 forward 完后，prompt 的 KV 散落在 P 的 KV pool 里。要把它们传给 D，步骤是：

  1. 算 kv_indices：从 req_to_token_pool 取该 req 的 KV slot 列表
  2. 转 page_indices：如果 page_size > 1，按 page 粒度聚合
  3. 塞进 FastQueue：sender.send() 把 chunk 加到异步队列
  4. transfer_worker 消费：调 libmooncake 的 send_kvcache
// prefill.py:744-822
def send_kv_chunk(self, req, last_chunk=False, end_idx=None):
    page_size = self.token_to_kv_pool_allocator.page_size
    start_idx = req.start_send_idx
    end_idx = end_idx or min(len(req.fill_ids), len(req.origin_input_ids))
    if not last_chunk:
        end_idx = end_idx - end_idx % page_size  # 页对齐

    kv_indices = (self.req_to_token_pool.req_to_token[req.req_pool_idx, start_idx:end_idx]
                  .cpu().numpy())
    req.start_send_idx = end_idx

    if last_chunk:
        self.disagg_metadata_buffers.set_buf(req)  # ★ 首 token/logprob 写进共享 buffer

    page_indices = kv_to_page_indices(kv_indices, page_size)
    req.disagg_kv_sender.send(page_indices, state_indices)
// mooncake/conn.py:1146-1343（transfer_worker 核心）
def transfer_worker(self, queue, executor):
    while True:
        chunk: TransferKVChunk = queue.get()
        reqs = self.transfer_infos[chunk.room].values()

        for req in reqs:
            target = self.decode_kv_args_table[req.mooncake_session_id]
            dst_slice = req.dst_kv_indices[chunk.index_slice]
            ret = self.send_kvcache(
                req.mooncake_session_id,
                chunk.prefill_kv_indices,
                target.dst_kv_ptrs,
                dst_slice,
                executor,
            )
            if chunk.is_last_chunk:
                self.send_aux(req, chunk.prefill_aux_index, target.dst_aux_ptrs)
                self.sync_status_to_decode_endpoint(req.endpoint, req.dst_port, req.room, Success)

PART 5P 的 event loop

// prefill.py:388-417
@torch.no_grad()
def event_loop_normal_disagg_prefill(self):
    while True:
        recv_reqs = self.recv_requests()                                 # 1
        self.process_input_requests(recv_reqs)                            # 2 分流到 BootstrapQueue
        self.waiting_queue.extend(                                          # 3 检查 bootstrap 状态
            self.disagg_prefill_bootstrap_queue.pop_bootstrapped()
        )
        batch = self.get_next_disagg_prefill_batch_to_run()              # 4 PrefillAdder 装箱
        self.cur_batch = batch
        if batch:
            result = self.run_batch(batch)                                 # 5 forward
            self.process_batch_result(batch, result)                       # 6 dispatch 到 disagg 分支
        else:
            self.on_idle()
        self.process_disagg_prefill_inflight_queue()                      # 7 清理 done
        self.last_batch = batch

和 unified 相比只多了第 3、7 步，run_batch/process_batch_result 基本共用，只是 disagg 分支里替换了"流式发 token"为"通过 metadata buffer 捎带首 token"。

PART 6本章要点

问题答案
P 有几个队列？3 个：BootstrapQueue → waiting_queue → InflightQueue
P 怎么知道可以 forward 了？后台 bootstrap_thread 把 KVPoll 改成 WaitingForInput；主循环 pop_bootstrapped 检测到后挪队
KV 什么时候开始传？run_batch 结束后，process_batch_result_disagg_prefill 调 send_kv_chunk，进 FastQueue；transfer_worker 池异步消费
首 token 怎么传到 D？send_kv_chunk 最后一段时，先 set_buf 把 token 写进 metadata buffer；然后 transfer_worker 先传 KV，再传 aux（metadata buffer 的那一个 slot）
为什么需要 InflightQueue？forward 完不等于 KV 传完。InflightQueue 专门等 RDMA 结束，成功后才释放 P 的 KV 和 tree cache 前缀锁
