PART 1Decode Worker 的核心数据结构

D 是 PD 架构里最复杂的一侧。因为它既要 在 KV 还没到位之前提前占好 KV 空间，又要避免这些提前占用挤爆真正在跑 decode 的 running_batch。

DecodeReqToTokenPool：关键魔改

// disaggregation/decode.py:96-106（注释）
"""
普通 ReqToTokenPool:
  #pre-allocated + #transfer + #running <= max_running_requests

DecodeReqToTokenPool:
  #running <= max_running_requests
  #pre-allocated + #transfer <= pre_alloc_size
"""
为什么要改：如果沿用普通 pool，那些"已经预分配了 KV 但还在等 P 传数据"的请求会占用 max_running_requests 额度。当 KV transfer 慢时，running_batch 的坑位都被等 KV 的请求占走，真正能 decode 的请求数急剧下降。DecodeReqToTokenPool 把这两类请求放到独立预算里。
🧠 DecodeReqToTokenPool 分区可视化
对比普通 pool vs DecodeReqToTokenPool：同样的请求流入，两者表现完全不同
❌ 普通 ReqToTokenPool（max = 8）
running 额度剩余: 8 可接受新请求: 8 decode tokens/s: 0
✅ DecodeReqToTokenPool（max_running=8 + pre_alloc=8）
running:
pre-alloc / transfer:
running 额度剩余: 8 pre-alloc 额度剩余: 8 decode tokens/s: 0

PART 2深读 1：DecodePreallocQueue.add

// disaggregation/decode.py:406-462
def add(self, req: Req, is_retracted: bool = False) -> None:
    if self._check_if_req_exceed_kv_capacity(req):
        return

    if is_retracted:
        self.retracted_queue.append(req)
        return

    decode_req = self._create_receiver_and_enqueue(req)

    if _is_fake_transfer(req, self.scheduler.server_args):
        decode_req.kv_receiver.init(0); return

    # 快速路径：dp_rank 已在 cache 里
    prefill_dp_rank = self._resolve_prefill_dp_rank(req)
    if prefill_dp_rank is not None:
        decode_req.kv_receiver.init(prefill_dp_rank)
        return

    # 慢路径：进 pending，下一轮集体查 bootstrap server
    self.pending_reqs.append(decode_req)

def _create_receiver_and_enqueue(self, req):
    kv_receiver = MooncakeKVReceiver(
        mgr=self.kv_manager,
        bootstrap_addr=f"{req.bootstrap_host}:{req.bootstrap_port}",
        bootstrap_room=req.bootstrap_room,
    )
    decode_req = DecodeRequest(req=req, kv_receiver=kv_receiver)
    self.queue.append(decode_req)
    return decode_req
注意：这一步只是登记请求，没有分配 KV page。真正分配发生在下一轮 pop_preallocated，那里会检查 pool 余量再决定。

PART 3深读 2：pop_preallocated + send_metadata

这是 D 侧最精妙的一步：在 P 还没开始算的时候，D 先给这个请求在自己的 KV pool 里挖好坑，并把坑的地址告诉 P。

// decode.py:645-817（精简）
def pop_preallocated(self, rids_to_check=None):
    self._resolve_pending_reqs()   # 解析慢路径的 dp_rank
    self._update_handshake_waiters()

    retractable_tokens = sum(len(r.origin_input_ids)+len(r.output_ids)
                             for r in self.scheduler.running_batch.reqs)
    allocatable = self._allocatable_tokens(retractable_tokens, count_retracted=True)

    for i, decode_req in enumerate(self.queue):
        if not decode_req.waiting_for_input: continue
        if self.req_to_token_pool.available_size() <= 0: break
        if self.req_to_metadata_buffer_idx_allocator.available_size() <= 0: break

        # 内存预算检查
        origin_len = len(decode_req.req.origin_input_ids)
        needed = origin_len + self.num_reserved_decode_tokens
        if needed > allocatable: break
        allocatable -= needed

        # ★ 真正分配 KV page
        dst_kv_indices = self._pre_alloc(decode_req.req)
        kv_indices = (self.req_to_token_pool.req_to_token[decode_req.req.req_pool_idx]
                       [:origin_len].cpu().numpy())
        page_indices = kv_to_page_indices(kv_indices, page_size)

        decode_req.metadata_buffer_index = self.req_to_metadata_buffer_idx_allocator.alloc()
        # ★ 把 page 地址发给 P
        decode_req.kv_receiver.send_metadata(
            page_indices, decode_req.metadata_buffer_index, state_indices,
        )
        preallocated.append(decode_req)
// mooncake/conn.py:1820-1867
def send_metadata(self, kv_indices, aux_index, state_indices):
    for bootstrap_info in self.bootstrap_infos:
        sock, lock = self._connect_to_bootstrap_server(bootstrap_info)
        with lock:
            sock.send_multipart([
                str(self.bootstrap_room).encode(),   # room
                self.kv_mgr.local_ip.encode(),        # D endpoint for reverse ZMQ
                str(self.kv_mgr.rank_port).encode(),
                self.session_id.encode(),             # mooncake session
                kv_indices.tobytes(),                 # ★ 目标 page 索引
                str(aux_index).encode(),              # metadata buffer slot
                state_indices_bytes,
                str(self.required_dst_info_num).encode(),
            ])

PART 4深读 3：DecodeTransferQueue.pop_transferred + commit

P 把 KV 和 aux（metadata）都传过来之后，D 侧的 DecodeTransferQueue 负责"收货"——commit 到 req 上。

// decode.py:946-1168（精简）
def pop_transferred(self):
    polls = poll_and_all_reduce([dr.kv_receiver for dr in self.queue])

    for dr, poll in zip(self.queue, polls):
        if poll == KVPoll.Failed:
            abort; release_kv_cache(dr); continue
        elif poll == KVPoll.Success:
            should_remove = self._commit_transfer_to_req(dr)
            if should_remove:
                transferred.append(dr.req)

def _commit_transfer_to_req(self, dr) -> bool:
    idx = dr.metadata_buffer_index
    (output_id, cached_tokens, logprobs_val, ...,
     output_bootstrap_room) = self.metadata_buffers.get_buf(idx)

    actual_room = output_bootstrap_room[0].item()
    expected = dr.req.bootstrap_room or 0

    if actual_room == 0: return False  # metadata 还没刷到
    elif actual_room != expected:
        # ★★ metadata buffer 串包，立即 abort
        prepare_abort(dr.req, "Metadata corruption detected...")
        dr.kv_receiver.clear()
        return True

    # 正常提交
    dr.req.output_ids.append(output_id[0].item())
    dr.req.cached_tokens = cached_tokens[0].item()
    if dr.req.return_logprob:
        dr.req.output_token_logprobs_val.append(logprobs_val[0].item())
    return True
bootstrap_room 校验是 PD 一致性的最后一道防线。metadata_buffer 是一个共享池（size 个 slot 循环用），如果 buffer 分配/释放时序错乱，可能拿到别人的 metadata。这里强校验才提交。

@torch.no_grad()
def event_loop_normal_disagg_decode(self):
    while True:
        recv_reqs = self.recv_requests()
        self.process_input_requests(recv_reqs)
        self.process_decode_queue()                      # ★ PD 专属

        batch = self.get_next_disagg_decode_batch_to_run()  # prebuilt extend batch
        self.cur_batch = batch
        if batch:
            result = self.run_batch(batch)
            self.process_batch_result(batch, result)
        else:
            self.on_idle()
// decode.py:1331-1362
def process_decode_queue(self):
    # 先恢复 retracted 请求
    resumed = self.disagg_decode_prealloc_queue.resume_retracted_reqs()
    self.waiting_queue.extend(resumed)
    if self.disagg_decode_prealloc_queue.retracted_queue: return

    # 降频 poll：避免每 iter 都 poll CPU 集合通信
    self.polling_count = (self.polling_count + 1) % self.polling_interval
    if self.polling_count % self.polling_interval == 0:
        req_conns, _ = self.disagg_decode_prealloc_queue.pop_preallocated()
        self.disagg_decode_transfer_queue.extend(req_conns)
        transferred = self.disagg_decode_transfer_queue.pop_transferred()
        self.waiting_queue.extend(transferred)
polling_interval 的意义：每 iter 跑 all_reduce poll 状态对 TP 组是个不小的 CPU 开销。D 侧用 polling_interval（默认 5）降频，只在每 N 轮检查一次。

PART 6本章要点

问题答案
D 有几个队列？3 个：PreallocQueue → TransferQueue → waiting_queue（+ running_batch）
为什么 D 要特殊的 ReqToTokenPool？隔离 pre-alloc 预算和 running 预算，避免 slow transfer 挤爆 running_batch
D 什么时候分配 KV？pop_preallocated 里，不是 add 的时候
send_metadata 发的是什么？(bootstrap_room, D 自己的 ZMQ endpoint, mooncake session, 目标 page 索引, aux 索引)
首 token 怎么到达 D？P 写进共享 metadata buffer → Mooncake send_aux RDMA 到 D → D 的 commit 从 buffer 读出来
bootstrap_room 校验防什么？metadata buffer slot 复用导致的串包（另一请求的 metadata 被误读）
