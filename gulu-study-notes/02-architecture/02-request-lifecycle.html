# tokenizer/tokenize.py:14
def tokenize(self, msgs):
    results = []
    for msg in msgs:
        if isinstance(msg.text, list):
            prompt = self.tokenizer.apply_chat_template(   # 对话格式化
                msg.text, tokenize=False, add_generation_prompt=True,
            )
        else:
            prompt = msg.text
        input_ids = self.tokenizer.encode(prompt, return_tensors="pt")
        results.append(input_ids.view(-1).to(torch.int32))
    return results

对于 "你好"，Qwen3 chat template 会展开成大约 20-30 个 token（含 system prompt + im_start/im_end 标记）。

# scheduler/cache.py:42
def allocate_paged(self, reqs):
    needed_pages = 0
    allocation_info = []
    for req in reqs:
        first_page = div_ceil(req.cached_len, self.page_size)   # 0
        last_page  = div_ceil(req.device_len,  self.page_size)   # 25
        if last_page > first_page:
            needed_pages += last_page - first_page                  # 25
            allocation_info.append((req.table_idx, first_page, last_page))
    if needed_pages > 0:
        allocated = self._page_to_token(self._allocate(needed_pages))
        _write_page_table(self.page_table, allocated, allocation_info, self.page_size)
page_table 是什么：一个 [max_running_req+1, max_seq_len] 的 GPU int32 tensor。page_table[req.table_idx, k] 存放的是"第 k 个 token 在 KV pool 里的物理位置"。它是 paged attention 的"地址映射表"。

# scheduler/scheduler.py:236  _make_positions
def _make_positions(batch, device):
    needed_size = sum(r.extend_len for r in batch.padded_reqs)   # 25
    indices_host = torch.empty(needed_size, dtype=torch.int32, pin_memory=True)
    offset = 0
    for req in batch.padded_reqs:
        torch.arange(req.cached_len, req.device_len,    # [0,1,2,...,24]
                     dtype=torch.int32,
                     out=indices_host[offset : offset + req.extend_len])
        offset += req.extend_len
    return indices_host.to(device, non_blocking=True)

三个 tuple 的语义：

tupleshape含义
positions[total_extend_len]每个 token 在序列中的位置（用于 RoPE）
input_tuple(table_idx_per_token, positions)用 advanced indexing 把 token 从 token_pool 取出
write_tuple(table_idx_per_req, last_pos_or_-1)下一个 token 写回 token_pool 的位置

engine.forward_batch · 真正的 GPU 计算
# engine/engine.py:191
def forward_batch(self, batch, args):
    assert torch.cuda.current_stream() == self.stream
    with self.ctx.forward_batch(batch):              # 把 batch 挂到全局 ctx
        if self.graph_runner.can_use_cuda_graph(batch):
            logits = self.graph_runner.replay(batch)
        else:
            logits = self.model.forward()              # prefill 走 eager

    for req in batch.reqs:
        req.complete_one()                               # cached_len ← device_len; device_len += 1

    next_tokens_gpu = self.sampler.sample(logits[:batch.size], args).to(torch.int32)
    next_tokens_cpu = next_tokens_gpu.to("cpu", non_blocking=True)   # D2H async
    copy_done_event = torch.cuda.Event()
    copy_done_event.record(self.stream)
    return ForwardOutput(next_tokens_gpu, next_tokens_cpu, copy_done_event)
三个返回值的妙处：next_tokens_gpu 留给下一步 decode（不离开 GPU），next_tokens_cpu 给 CPU 用来发给 detokenizer，copy_done_event 让 CPU 知道什么时候这个异步 D2H 拷贝完成。

Uvicorn 把每个 yield 出去的 chunk 通过 HTTP/1.1 chunked-encoding 直接 flush 到客户端。第一个 chunk 到达客户端的时刻就是 TTFT。

四、Decode 循环（步骤 ⑮~⑰）

Prefill 后这个 req 自动进入 decode_manager.running_reqs（通过 scheduler._forward 末尾的 filter_reqs）。下一轮 overlap_loop：

# scheduler.py:227  _forward
def _forward(self, forward_input):
    batch, sample_args, input_mapping, output_mapping = forward_input
    batch.input_ids = self.token_pool[input_mapping]
    forward_output = self.engine.forward_batch(batch, sample_args)
    self.token_pool[output_mapping] = forward_output.next_tokens_gpu  ★ GPU 直接写回
    self.decode_manager.filter_reqs(forward_input.batch.reqs)        # 加入 decode
    return forward_output

从此每步 _schedule_next_batch 走 decode 路径：

  • decode_manager.schedule_next_batch() 直接把 running_reqs 一锅端
  • graph_runner.can_use_cuda_graph(batch) 返回 True（decode + bs 在录制范围内）
  • graph_runner.replay(batch) 不重启 kernel launch，直接 replay 录制好的图
  • 每步只有 1 个新 token，extend_len = 1
第 5 个 token 出来后：req.can_decode 返回 False（remain_len = 0），_process_last_data 触发 finished=True： 
def _free_req_resources(self, req):
    self.table_manager.free(req.table_idx)         # 槽位回收
    self.cache_manager.cache_req(req, finished=True)  # 完整序列插入 Radix, 尾部释放
同时发送 DetokenizeMsg(finished=True)，detokenizer 出 UserReply(finished=True)，stream_chat_completions 看到后发 data: [DONE]，HTTP stream 关闭。

五、时间剖析（典型 H200 数值）

阶段耗时量级瓶颈
HTTP + ASGI 路由~100 µsFastAPI
ZMQ Push (api→tok)~30 µsIPC socket
tokenize 25 token~500 µsHF tokenizer Python 开销
ZMQ Push (tok→sched)~30 µsIPC
_schedule_next_batch + _prepare_batch~200 µsPython + 几个小 H2D copy
prefill forward (25 token, eager)~10-20 msGPU compute, 占 TTFT 主要部分
D2H copy + sync~50 µs1 个 int32
cache_req + Radix insert~80 µsPython tree walk
ZMQ + detokenize + ZMQ~200 µs2 个 IPC + 1 次 batch_decode
SSE flush~30 µsuvicorn
总 TTFT~12-22 msprefill GPU 占大头
每步 decode (CUDA Graph)~3-5 ms / tokenGPU mem-bound
每步 decode 全链路 (含 ZMQ)~3.5-5.5 ms / token得益于 overlap，CPU 部分被吃掉
Overlap Scheduling 怎么"省"出时间：第 ⑪⑫⑬⑭ 步（处理 last_data + detokenize + 回复 client）和第 ⑨⑩ 步（forward + sample）发生在同一次 loop iteration，但因为 forward 是异步 launch + GPU 执行，CPU 在等 GPU 时同步处理上一批的结果。理想情况下 CPU 路径完全被 GPU 时间覆盖，TTFT 和 ITL 都只剩 GPU 时间。
