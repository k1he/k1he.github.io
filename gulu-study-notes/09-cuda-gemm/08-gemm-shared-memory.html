C7 · GEMM Shared Memory · Tiled 实现 CUDA GEMM 系列 · 8/15

把 tile 从 global memory 搬到 shared memory，让 block 内 threads 共享数据——这一步从 1-5% cuBLAS 跳到 15-25%。对应 Triton T8-T9 自动做的事。

§1核心思路回顾

整个 kernel 的工作模式可以归纳为一个 两步搬运 循环：

for each K-block (k = 0, BK, 2*BK, ...):
  1. 所有 threads 协作: Global → Shared (A_tile + B_tile)
  2. __syncthreads()
  3. 每个 thread: Shared → Register 计算
  4. __syncthreads()
对比 Triton: Triton 的 tl.load 自动把数据搬到 shared memory（你看不到 __shared__）。这里我们手动做了完全相同的事。

§2完整 Kernel 代码

#define BM 32
#define BN 32
#define BK 32

__global__ void gemm_shared(
    const float* A, const float* B, float* C,
    int M, int N, int K
) {
    // Shared memory for tiles
    __shared__ float A_shared[BM][BK];
    __shared__ float B_shared[BK][BN];

    // Thread position within block
    int tx = threadIdx.x;  // 0..31 (col within block)
    int ty = threadIdx.y;  // 0..31 (row within block)

    // Output position
    int row = blockIdx.y * BM + ty;
    int col = blockIdx.x * BN + tx;

    float sum = 0.0f;

    // K-loop: iterate over tiles
    for (int k_tile = 0; k_tile < (K + BK - 1) / BK; k_tile++) {
        // Step 1: Cooperative loading to shared memory
        int a_col = k_tile * BK + tx;
        int b_row = k_tile * BK + ty;

        // Load A tile
        if (row < M && a_col < K)
            A_shared[ty][tx] = A[row * K + a_col];
        else
            A_shared[ty][tx] = 0.0f;

        // Load B tile
        if (b_row < K && col < N)
            B_shared[ty][tx] = B[b_row * N + col];
        else
            B_shared[ty][tx] = 0.0f;

        __syncthreads();  // Wait for all loads

        // Step 2: Compute using shared memory
        for (int kk = 0; kk < BK; kk++) {
            sum += A_shared[ty][kk] * B_shared[kk][tx];
        }

        __syncthreads();  // Wait before next load
    }

    // Write result
    if (row < M && col < N)
        C[row * N + col] = sum;
}

// Launch
dim3 block(BN, BM);  // (32, 32) = 1024 threads
dim3 grid((N + BN - 1) / BN, (M + BM - 1) / BM);
gemm_shared<<<grid, block>>>(d_A, d_B, d_C, M, N, K);

§3逐步解剖

3.1 Shared Memory 声明

__shared__ float A_shared[BM][BK];  // 32×32×4 = 4 KB
__shared__ float B_shared[BK][BN];  // 32×32×4 = 4 KB
// 总共 8 KB per block
项目值说明
A_shared 大小32 × 32 × 4B = 4 KB存 A 的一个 tile
B_shared 大小32 × 32 × 4B = 4 KB存 B 的一个 tile
每个 block 总用8 KB远低于 SM 的 48-164 KB 限制
最大并发 blocks48 KB / 8 KB = 6受 shared memory 容量限制
关键点：__shared__ 变量生命周期 = block 生命周期。同一 block 内所有 1024 个 threads 看到同一份数据。不同 block 有自己独立的副本。

3.2 协作加载 (Cooperative Loading)

Block (32, 32) = 1024 threads
Tile size = 32 × 32 = 1024 elements
→ 每个 thread 加载恰好 1 个元素! 
// Thread (ty, tx) 负责加载:
A_shared[ty][tx] = A[row][k_tile*BK + tx]
// 同一 warp (固定 ty, tx=0..31) 读 A 的连续地址 → coalesced ✓

B_shared[ty][tx] = B[k_tile*BK + ty][col]
// 同一 warp (固定 ty, tx=0..31) 读 B 的连续地址 → coalesced ✓
Coalesced access: 同一 warp 的 32 个 threads (tx = 0..31) 访问连续 32 个 float = 128 bytes = 恰好一次 128B transaction。完美合并！B 的加载同理。

3.3 内层计算循环

for (int kk = 0; kk < BK; kk++) {
    sum += A_shared[ty][kk] * B_shared[kk][tx];
}
// ty 固定 → 读 A_shared 的一整行 (32 elements)
// tx 固定 → 读 B_shared 的一整列 (32 elements)
// 全从 shared memory 读! ~20 cycles vs global 的 ~400 cycles
Bank conflict 检查：
• A_shared[ty][kk] — 同一 warp 不同 tx 读同一 kk（broadcast）→ 无 conflict ✓
• B_shared[kk][tx] — 同一 warp 不同 tx 读同一行的不同列 → 连续 bank → 无 conflict ✓

3.4 两次 __syncthreads() 的含义

第一次 sync (加载后)

"确保所有 thread 都加载完再开始计算"

如果不 sync：Thread A 可能读到 Thread B 还没写入的 shared memory 位置 → 脏数据

第二次 sync (计算后)

"确保所有 thread 都计算完再加载下一个 tile"

如果不 sync：Thread A 开始覆写下一个 tile 时，Thread B 可能还在读当前 tile → 数据覆盖

§4为什么快了？— 数据复用量化

Naive（无 shared memory）

一个 thread 读: K 个 A 元素 + K 个 B 元素 (全从 global)
整个 block 的 1024 threads 总读: 1024 × 2K = 2048K from Global Memory

Shared Memory Tiling（本节）

一个 K-tile 加载: 32×32 + 32×32 = 2048 元素 from global
K/32 个 tiles → 总读: (K/32) × 2048 = 64K from Global Memory

节省比

2048K / 64K = 32× = BK 倍!
直觉：数据从 global 被每个 block 的 threads 共享了 32 次（= BK 次）。每个 A 元素被同一行的 32 个 threads 读取，每个 B 元素被同一列的 32 个 threads 读取。这就是 tiling 的数据复用。
对比 Triton：Triton 的 tl.load 自动把数据放进 shared memory（你看不到）。这里我们手动做了完全相同的事。理解了这里，你就理解了 Triton 在背后做什么。

§6还差什么？

问题原因解决方案预期提升
Thread tile 太小 每个 thread 只算 1 个 C 元素
→ register 没被充分利用		 让每个 thread 算 TM×TN 个 C 元素
(register tiling)		 → 40-60% cuBLAS
没有 double buffering Load 和 compute 串行：
load → sync → compute → sync → ...		 在计算当前 tile 时预取下一个 tile
(overlap load/compute)		 → 60-80% cuBLAS
没用 Tensor Core 用 FMA 指令 (1 FLOP/cycle/core)

§7完整可运行代码

// gemm_shared.cu
// 编译: nvcc -O3 -o gemm_shared gemm_shared.cu
// 运行: ./gemm_shared

#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <cuda_runtime.h>

#define BM 32
#define BN 32
#define BK 32

#define CUDA_CHECK(call) do { \
    cudaError_t err = call; \
    if (err != cudaSuccess) { \
        printf("CUDA error: %s at %s:%d\n", \
               cudaGetErrorString(err), __FILE__, __LINE__); \
        exit(1); \
    } \
} while(0)

// ============= Shared Memory GEMM =============
__global__ void gemm_shared(
    const float* A, const float* B, float* C,
    int M, int N, int K
) {
    __shared__ float A_shared[BM][BK];
    __shared__ float B_shared[BK][BN];

    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int row = blockIdx.y * BM + ty;
    int col = blockIdx.x * BN + tx;

    float sum = 0.0f;

    for (int k_tile = 0; k_tile < (K + BK - 1) / BK; k_tile++) {
        int a_col = k_tile * BK + tx;
        int b_row = k_tile * BK + ty;

        A_shared[ty][tx] = (row < M && a_col < K)
                           ? A[row * K + a_col] : 0.0f;
        B_shared[ty][tx] = (b_row < K && col < N)
                           ? B[b_row * N + col] : 0.0f;

        __syncthreads();

        for (int kk = 0; kk < BK; kk++)
            sum += A_shared[ty][kk] * B_shared[kk][tx];

        __syncthreads();
    }

    if (row < M && col < N)
        C[row * N + col] = sum;
}

// ============= Naive GEMM (对照组) =============
__global__ void gemm_naive(
    const float* A, const float* B, float* C,
    int M, int N, int K
) {
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;
    if (row < M && col < N) {
        float sum = 0.0f;
        for (int k = 0; k < K; k++)
            sum += A[row * K + k] * B[k * N + col];
        C[row * N + col] = sum;
    }
}

// ============= Host =============
int main() {
    const int M = 1024, N = 1024, K = 1024;
    size_t sA = M * K * sizeof(float);
    size_t sB = K * N * sizeof(float);
    size_t sC = M * N * sizeof(float);

    // Host alloc
    float* h_A = (float*)malloc(sA);
    float* h_B = (float*)malloc(sB);
    float* h_C_naive  = (float*)malloc(sC);
    float* h_C_shared = (float*)malloc(sC);

    srand(42);
    for (int i = 0; i < M * K; i++) h_A[i] = (float)rand() / RAND_MAX;
    for (int i = 0; i < K * N; i++) h_B[i] = (float)rand() / RAND_MAX;

    // Device alloc
    float *d_A, *d_B, *d_C;
    CUDA_CHECK(cudaMalloc(&d_A, sA));
    CUDA_CHECK(cudaMalloc(&d_B, sB));
    CUDA_CHECK(cudaMalloc(&d_C, sC));
    CUDA_CHECK(cudaMemcpy(d_A, h_A, sA, cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(d_B, h_B, sB, cudaMemcpyHostToDevice));

    dim3 block(BN, BM);  // (32,32) = 1024 threads
    dim3 grid((N + BN - 1) / BN, (M + BM - 1) / BM);

    cudaEvent_t t0, t1;
    cudaEventCreate(&t0);
    cudaEventCreate(&t1);
    const int WARMUP = 5, ITERS = 50;
    float ms;
    double flops = 2.0 * M * N * K;

    // --- Benchmark Naive ---
    for (int i = 0; i < WARMUP; i++)
        gemm_naive<<<grid, block>>>(d_A, d_B, d_C, M, N, K);
    cudaEventRecord(t0);
    for (int i = 0; i < ITERS; i++)
        gemm_naive<<<grid, block>>>(d_A, d_B, d_C, M, N, K);
    cudaEventRecord(t1);
    cudaEventSynchronize(t1);
    cudaEventElapsedTime(&ms, t0, t1);
    ms /= ITERS;
    printf("Naive:  %7.3f ms  %.2f TFLOPS\n", ms, flops/(ms*1e9));
    CUDA_CHECK(cudaMemcpy(h_C_naive, d_C, sC, cudaMemcpyDeviceToHost));

    // --- Benchmark Shared ---
    for (int i = 0; i < WARMUP; i++)
        gemm_shared<<<grid, block>>>(d_A, d_B, d_C, M, N, K);
    cudaEventRecord(t0);
    for (int i = 0; i < ITERS; i++)
        gemm_shared<<<grid, block>>>(d_A, d_B, d_C, M, N, K);
    cudaEventRecord(t1);
    cudaEventSynchronize(t1);
    cudaEventElapsedTime(&ms, t0, t1);
    ms /= ITERS;
    printf("Shared: %7.3f ms  %.2f TFLOPS\n", ms, flops/(ms*1e9));
    CUDA_CHECK(cudaMemcpy(h_C_shared, d_C, sC, cudaMemcpyDeviceToHost));

    // --- Verify ---
    float max_err = 0.0f;
    for (int i = 0; i < M * N; i++) {
        float err = fabsf(h_C_naive[i] - h_C_shared[i]);
        if (err > max_err) max_err = err;
    }
    printf("Max error vs naive: %e  %s\n",
           max_err, max_err < 1e-3 ? "PASS" : "FAIL");

    // Cleanup
    free(h_A); free(h_B);
    free(h_C_naive); free(h_C_shared);
    cudaFree(d_A); cudaFree(d_B); cudaFree(d_C);
    cudaEventDestroy(t0); cudaEventDestroy(t1);
    return 0;
}

// 编译: nvcc -O3 -o gemm_shared gemm_shared.cu
// 运行: ./gemm_shared
预期输出 (A100)：

Shared: 0.085 ms 25.3 TFLOPS
Max error vs naive: 1.5e-05 PASS
Shared 比 Naive 快约 10-12x。

§8带走

#要点关键词
1Shared memory tiling: Global → Shared → Register 两步搬运数据层级
2协作加载: block 内所有 threads 分工加载，然后共享使用cooperative load
3两次 __syncthreads: 加载完才能计算，计算完才能加载下一个barrier
4数据复用 BK 倍 → 从 HBM 总读取量下降 BK 倍reuse factor
5性能 ~15-20% cuBLAS（还有 register tiling + Tensor Core 没做）optimization roadmap
下一节预告（C8）：Register Tiling — 让每个 thread 算 TM×TN 个 C 元素，充分利用 register file，从 20% 跳到 40-60% cuBLAS。
