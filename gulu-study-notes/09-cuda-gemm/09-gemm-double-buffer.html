C8 · GEMM Double Buffering · 重叠加载与计算 CUDA GEMM 系列 · 9/15

在计算当前 tile 的同时，预取下一个 tile——让 memory latency 被计算完全覆盖。这对应 Triton 的 num_stages 参数。

§1问题：Load 和 Compute 的串行化

没有 double buffering 时，load 和 compute 必须严格交替进行：

Timeline (没有 double buffering):
Load tile 0 → ███████
__sync
Compute 0 → ███████
__sync
Load tile 1 → ███████
__sync
Compute 1 → ███████

Memory unit: ███ idle ███ idle
Compute unit: idle ███ idle ███
一半的时间 compute unit 在等数据，另一半时间 memory unit 在等计算完成。硬件利用率只有 ~50%。

§2解决：Double Buffering (Prefetch)

核心思想：用两组 shared memory buffer——一组在被计算使用，另一组在被加载。轮流切换。

Timeline (有 double buffering):
Prefetch tile 1: ███████████████
__sync
Compute 0 + Prefetch 2: ████████████████
__sync
Compute 1 + Prefetch 3: ████████████████
...

Memory: ████████████████████████████████
Compute: ████████████████████████████████
→ Fully overlapped!
对应 Triton 的 num_stages 参数：num_stages=2 就是 double buffering，num_stages=3 就是 triple buffering。

Triton 对应代码

# Triton 的 num_stages=2 就是 double buffering
# num_stages=3 就是 triple buffering
triton.Config({'BLOCK_M': 128, ...}, num_stages=2, ...)

§3实现方式 1：两组 Shared Memory

__shared__ float A_shared[2][BM][BK];  // 两个 buffer
__shared__ float B_shared[2][BK][BN];

int buf_read = 0;   // 当前正在读(计算用)的 buffer
int buf_write = 1;  // 当前正在写(加载用)的 buffer

// Prefetch first tile
load_tile(A_shared[buf_write], B_shared[buf_write], k=0);
__syncthreads();
swap(buf_read, buf_write);

for (int k_tile = 1; k_tile < num_k_tiles; k_tile++) {
    // 同时做两件事:
    // 1. 加载下一个 tile 到 buf_write
    load_tile(A_shared[buf_write], B_shared[buf_write], k_tile);

    // 2. 用 buf_read 中的当前 tile 做计算
    compute_tile(A_shared[buf_read], B_shared[buf_read], acc);

    __syncthreads();
    swap(buf_read, buf_write);
}
// 计算最后一个 tile
compute_tile(A_shared[buf_read], B_shared[buf_read], acc);
关键点：load_tile 和 compute_tile 操作的是不同的 buffer，所以它们不冲突。__syncthreads() 确保加载完成后再 swap。

§4实现方式 2：cp.async (Ampere+)

#include <cuda/pipeline>

// cp.async: 异步拷贝 global → shared，不占用计算资源
__pipeline_memcpy_async(&A_shared[buf][ty][tx], &A[global_idx], sizeof(float));
__pipeline_commit();

// ... do computation on other buffer ...

__pipeline_wait_prior(0);  // 等待异步拷贝完成
__syncthreads();

cp.async 是 Ampere (sm_80+) 的硬件特性：DMA engine 独立于 compute pipeline 搬运数据。

cp.async 的优势

特性说明
不占计算线程DMA engine 独立搬运，compute pipeline 完全空闲给计算
真正硬件级 overlap不是软件模拟，而是硬件同时执行 load 和 compute
Triton 自动使用当 num_stages ≥ 2 且目标 GPU ≥ sm_80 时，Triton 自动生成 cp.async
绕过 L1 cache直接 global → shared，避免污染 L1

§5完整 Double Buffer Kernel

综合 double buffering + register tiling 的完整 kernel：

#define BM 64
#define BN 64
#define BK 16
#define TM 8
#define TN 8

__global__ void gemm_double_buffer(
    const float* __restrict__ A,
    const float* __restrict__ B,
    float* __restrict__ C,
    int M, int N, int K
) {
    __shared__ float A_shared[2][BM][BK];
    __shared__ float B_shared[2][BK][BN];

    // Thread tile (每个 thread 算 TM×TN 个输出)
    float acc[TM][TN] = {0.0f};
    float a_frag[TM], b_frag[TN];

    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int bx = blockIdx.x;
    int by = blockIdx.y;

    // 每个 thread 在 tile 加载中负责的元素
    int tid = ty * blockDim.x + tx;
    int num_threads = blockDim.x * blockDim.y;

    int buf = 0;

    // ═══════════════════════════════════════════════
    // Prefetch first tile into buf=0
    // ═══════════════════════════════════════════════
    for (int i = tid; i < BM * BK; i += num_threads) {
        int row = i / BK;
        int col = i % BK;
        int global_row = by * BM + row;
        int global_col = col;  // k_tile=0, offset=0
        if (global_row < M && global_col < K)
            A_shared[0][row][col] = A[global_row * K + global_col];
        else
            A_shared[0][row][col] = 0.0f;
    }
    for (int i = tid; i < BK * BN; i += num_threads) {
        int row = i / BN;
        int col = i % BN;
        int global_row = row;  // k_tile=0, offset=0
        int global_col = bx * BN + col;
        if (global_row < K && global_col < N)
            B_shared[0][row][col] = B[global_row * N + global_col];
        else
            B_shared[0][row][col] = 0.0f;
    }
    __syncthreads();

    // ═══════════════════════════════════════════════
    // Main loop: compute buf, prefetch buf^1
    // ═══════════════════════════════════════════════
    int num_k_tiles = (K + BK - 1) / BK;

    for (int k_tile = 1; k_tile < num_k_tiles; k_tile++) {
        // ─── Prefetch next tile to buf^1 ───
        for (int i = tid; i < BM * BK; i += num_threads) {
            int row = i / BK;
            int col = i % BK;
            int global_row = by * BM + row;
            int global_col = k_tile * BK + col;
            if (global_row < M && global_col < K)
                A_shared[buf^1][row][col] = A[global_row * K + global_col];
            else
                A_shared[buf^1][row][col] = 0.0f;
        }
        for (int i = tid; i < BK * BN; i += num_threads) {
            int row = i / BN;
            int col = i % BN;
            int global_row = k_tile * BK + row;
            int global_col = bx * BN + col;
            if (global_row < K && global_col < N)
                B_shared[buf^1][row][col] = B[global_row * N + global_col];
            else
                B_shared[buf^1][row][col] = 0.0f;
        }

        // ─── Compute current tile from buf ───
        for (int kk = 0; kk < BK; kk++) {
            // Load fragments from shared memory
            for (int tm = 0; tm < TM; tm++)
                a_frag[tm] = A_shared[buf][ty * TM + tm][kk];
            for (int tn = 0; tn < TN; tn++)
                b_frag[tn] = B_shared[buf][kk][tx * TN + tn];

            // Outer product
            for (int tm = 0; tm < TM; tm++)
                for (int tn = 0; tn < TN; tn++)
                    acc[tm][tn] += a_frag[tm] * b_frag[tn];
        }

        __syncthreads();
        buf ^= 1;  // swap buffers
    }

    // ═══════════════════════════════════════════════
    // Compute last tile (already in buf)
    // ═══════════════════════════════════════════════
    for (int kk = 0; kk < BK; kk++) {
        for (int tm = 0; tm < TM; tm++)
            a_frag[tm] = A_shared[buf][ty * TM + tm][kk];
        for (int tn = 0; tn < TN; tn++)
            b_frag[tn] = B_shared[buf][kk][tx * TN + tn];

        for (int tm = 0; tm < TM; tm++)
            for (int tn = 0; tn < TN; tn++)
                acc[tm][tn] += a_frag[tm] * b_frag[tn];
    }

    // ═══════════════════════════════════════════════
    // Write back to global memory
    // ═══════════════════════════════════════════════
    for (int tm = 0; tm < TM; tm++)
        for (int tn = 0; tn < TN; tn++) {
            int row = by * BM + ty * TM + tm;
            int col = bx * BN + tx * TN + tn;
            if (row < M && col < N)
                C[row * N + col] = acc[tm][tn];
        }
}
启动参数：dim3 block(BN/TN, BM/TM) = (8, 8) = 64 threads，dim3 grid(N/BN, M/BM)。每个 thread 计算 8×8 = 64 个 C 元素。

§6Register Tiling (Thread Tile) 解释

注意这里每个 thread 不再只算 1 个 C 元素，而是 TM×TN = 8×8 = 64 个！

为什么要 Register Tiling？

方案每 thread 算多少Block 线程数Register 使用
1 element/thread164×64 = 4096 (超限!)几乎没有复用
64 elements/thread8×8 = 648×8 = 64 threads数据在 register 中被充分复用
Block threads: (BM/TM) × (BN/TN) = 8 × 8 = 64 threads

每个 thread 持有:
  acc[8][8] = 64 registers (累加器)
  a_frag[8] + b_frag[8] = 16 registers (数据片段)
  合计: ~80 registers/thread → OK (硬件上限 255)
核心优势：一次从 shared memory 加载 a_frag[8]，它被用于与 b_frag 的 8 个元素逐一相乘。即 1 次 shared memory 读 → 8 次乘加。这就是 register 复用。

§8Triple Buffering (num_stages=3)

Double buffering 的自然扩展——用 3 组 buffer 进一步覆盖长延迟：

角色Buffer 0Buffer 1Buffer 2
当前迭代ComputingLoadingLoaded & waiting
下一迭代Loaded & waitingComputingLoading
  • 更好地覆盖高延迟（HBM 延迟 ~400 cycles）
  • 代价：需要 3× shared memory（BM×BK×3 + BK×BN×3）
  • Triton: num_stages=3 自动生成三重缓冲 + cp.async pipeline
  • 实际中 num_stages=2~4 最常用，再多 shared memory 不够
