C3 · Vector Add · 第一个完整 CUDA Kernel CUDA GEMM 系列 · 4/15

从头到尾写一个 CUDA vector add：分配内存、传输数据、启动 kernel、验证结果。对比 Triton 版，感受 CUDA "什么都要自己管" 的风格。

§1完整流程概览

CUDA 编程流水线 — 5 步缺一不可：

1
cudaMalloc — 分配 GPU 内存
2
cudaMemcpy H→D — 把数据从 CPU 拷到 GPU
3
kernel<<<grid, block>>>(...) — 启动计算
4
cudaMemcpy D→H — 把结果拷回 CPU
5
cudaFree — 释放 GPU 内存
对比 Triton / PyTorch：
PyTorch 的 torch.rand(n, device='cuda') 帮你做了 1 + 2。Triton 的 kernel[grid](x, y, out, ...) 帮你做了 3。CUDA 全部手动。

§2Kernel 代码

__global__ void vector_add_kernel(const float* a, const float* b, float* c, int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        c[idx] = a[idx] + b[idx];
    }
}

逐行解释

代码含义
__global__标记为 kernel — CPU 调用，GPU 执行
const float*输入指针（指向 GPU 地址空间，非 CPU 地址）
blockIdx.x * blockDim.x + threadIdx.x全局线程编号（上一节学的索引公式）
if (idx < n)边界检查 — 等同于 Triton 的 mask = offs < n
c[idx] = a[idx] + b[idx]每个线程处理一个元素的加法

§3Host 端完整代码

可直接编译运行的完整程序：

// vec_add.cu — 第一个完整 CUDA 程序
#include <cuda_runtime.h>
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

// 错误检查宏 — 每个 CUDA API 调用都要包裹
#define CHECK_CUDA(call) do { \
    cudaError_t err = call; \
    if (err != cudaSuccess) { \
        fprintf(stderr, "CUDA error at %s:%d: %s\n", __FILE__, __LINE__, \
                cudaGetErrorString(err)); \
        exit(1); \
    } \
} while(0)

__global__ void vector_add(const float* a, const float* b, float* c, int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        c[idx] = a[idx] + b[idx];
    }
}

int main() {
    const int N = 1 << 20;  // 1M elements
    size_t bytes = N * sizeof(float);

    // ① Host 分配 + 初始化
    float *h_a = (float*)malloc(bytes);
    float *h_b = (float*)malloc(bytes);
    float *h_c = (float*)malloc(bytes);
    for (int i = 0; i < N; i++) {
        h_a[i] = sinf(i) * sinf(i);
        h_b[i] = cosf(i) * cosf(i);
    }

    // ② Device 分配
    float *d_a, *d_b, *d_c;
    CHECK_CUDA(cudaMalloc(&d_a, bytes));
    CHECK_CUDA(cudaMalloc(&d_b, bytes));
    CHECK_CUDA(cudaMalloc(&d_c, bytes));

    // ③ H → D 数据传输
    CHECK_CUDA(cudaMemcpy(d_a, h_a, bytes, cudaMemcpyHostToDevice));
    CHECK_CUDA(cudaMemcpy(d_b, h_b, bytes, cudaMemcpyHostToDevice));

    // ④ 启动 Kernel
    int block_size = 256;
    int grid_size = (N + block_size - 1) / block_size;
    vector_add<<<grid_size, block_size>>>(d_a, d_b, d_c, N);
    CHECK_CUDA(cudaGetLastError());

    // ⑤ D → H 结果拷回
    CHECK_CUDA(cudaMemcpy(h_c, d_c, bytes, cudaMemcpyDeviceToHost));

    // ⑥ 验证正确性
    float max_err = 0;
    for (int i = 0; i < N; i++) {
        max_err = fmaxf(max_err, fabsf(h_c[i] - (h_a[i] + h_b[i])));
    }
    printf("Max error: %e\n", max_err);
    printf(max_err < 1e-5 ? "PASS\n" : "FAIL\n");

    // ⑦ 释放内存
    cudaFree(d_a); cudaFree(d_b); cudaFree(d_c);
    free(h_a); free(h_b); free(h_c);
    return 0;
}
编译运行：nvcc -o vec_add vec_add.cu && ./vec_add
预期输出：Max error: 0.000000e+00 + PASS（sin² + cos² = 1，浮点精确）

§4Triton vs CUDA 对比

Triton (~15 行)

@triton.jit
def add_kernel(x_ptr, y_ptr, out_ptr, n,
               BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n
    x = tl.load(x_ptr + offs, mask=mask)
    y = tl.load(y_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, x + y, mask=mask)

# 启动 — 内存由 PyTorch 管理
x = torch.rand(N, device='cuda')
y = torch.rand(N, device='cuda')
out = torch.empty_like(x)
grid = lambda m: (triton.cdiv(N, m['BLOCK']),)
add_kernel[grid](x, y, out, N, BLOCK=1024)

CUDA (~50 行)

// 需要手动做的事情：
float *h_a = malloc(bytes);       // Host 分配
float *d_a;
cudaMalloc(&d_a, bytes);          // Device 分配
cudaMemcpy(d_a, h_a, ...H2D);    // 手动传输

// 手动计算 grid/block
int block = 256;
int grid = (N + block - 1) / block;
kernel<<<grid, block>>>(...);

cudaMemcpy(h_c, d_c, ...D2H);    // 手动拷回
cudaFree(d_a);                    // 手动释放
free(h_a);                        // Host 也要释放

// 还要检查每个调用的错误...
CHECK_CUDA(cudaMalloc(...));
CHECK_CUDA(cudaMemcpy(...));
CHECK_CUDA(cudaGetLastError());
维度TritonCUDA
内存管理PyTorch 代劳全部手动 malloc/free
数据传输自动（tensor 创建在 cuda）手动 cudaMemcpy 两次
错误处理Python 异常每行都要 CHECK_CUDA
Block size编译期常量，自动调优手动选，手动算 grid
代码量~15 行~50 行
控制粒度Block 级Thread 级（完全控制）
总结：这就是为什么先学 Triton — 它让你专注于算法。但 CUDA 给你完全控制权，在需要极致优化时不可替代。

§5CUDA Error Handling

方式 1：检查每个 API 调用

cudaError_t err = cudaMalloc(&ptr, size);
if (err != cudaSuccess) {
    fprintf(stderr, "Error: %s\n", cudaGetErrorString(err));
    exit(1);
}

方式 2：检查 Kernel Launch（异步！）

// Kernel 启动是异步的 — 不会立即报告执行错误
kernel<<<grid, block>>>(...);

// 检查 launch 配置错误（如 block 太大）
cudaError_t err = cudaGetLastError();

// 等 kernel 真正执行完
cudaDeviceSynchronize();

// 检查 execution 错误（如越界访问）
err = cudaGetLastError();
注意：Kernel 启动是异步的！kernel<<<>>>() 立即返回，不等 kernel 执行完。要用 cudaDeviceSynchronize() 或 cudaMemcpy（隐式同步）来等待。如果不等待就检查错误，可能漏掉 kernel 内部的崩溃。

§6性能计时

cudaEvent_t start, stop;
cudaEventCreate(&start);
cudaEventCreate(&stop);

cudaEventRecord(start);
vector_add<<<grid_size, block_size>>>(d_a, d_b, d_c, N);
cudaEventRecord(stop);
cudaEventSynchronize(stop);

float ms = 0;
cudaEventElapsedTime(&ms, start, stop);
printf("Kernel time: %.3f ms\n", ms);

// 带宽计算：读 a + 读 b + 写 c = 3N 次 float 访存
float gb = 3.0f * N * sizeof(float) / 1e9;
printf("Effective BW: %.1f GB/s\n", gb / (ms / 1000));
为什么用 cudaEvent 而不是 CPU 计时器？
因为 kernel 启动是异步的。CPU 的 clock() 只能测到 launch 开销（微秒级），测不到 kernel 真正执行时间。cudaEvent 是 GPU 时间戳，精度 ~0.5 微秒。
预期结果：Vector add 的理论带宽利用率应该接近 100% — 因为它是纯 memory-bound 的（计算：访存 = 1:12 bytes）。在 A100 上预期 ~1.5 TB/s。

§7编译选项

# 基本编译
nvcc -o vec_add vec_add.cu

# 指定 GPU 架构 (A100 = sm_80, H100 = sm_90, H20 = sm_89)
nvcc -arch=sm_80 -o vec_add vec_add.cu

# 优化等级（生产用 -O3）
nvcc -O3 -arch=sm_80 -o vec_add vec_add.cu

# 调试模式（-g: host debug, -G: device debug）
nvcc -g -G -o vec_add_debug vec_add.cu
选项含义场景
-arch=sm_XX指定目标 GPU 架构确保生成对应指令集
-O3最高优化性能测试 / 生产
-g -G开启 host + device 调试信息配合 cuda-gdb 使用
-lineinfo保留行号（不影响优化）profiler 定位热点
--ptxas-options=-v打印寄存器 / shared mem 使用量占用率分析

§8带走

  1. CUDA 完整流程：malloc → memcpy H2D → kernel<<<>>> → memcpy D2H → free
  2. Kernel = __global__ void 函数，每个 thread 执行一遍函数体
  3. 边界检查 if (idx < n) 等同于 Triton 的 mask
  4. Kernel 启动是异步的 — 需要 cudaDeviceSynchronize() 或隐式同步来等待
  5. 用 cudaEvent 计时 — 不要用 CPU 计时器测 kernel 性能
